> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 `judge.py` 从「记录单 seed 分数」改成「标定噪声、冻结尺子、强制复核」，让一夜搜索必须落地判定而不只是一串提交。

**Architecture:** 新增 `task.py` 持有数据根与评测入口并被冻结；`judge.py` 新增 `calibrate` 子命令实测 seed 方差，由它导出 `trial` 的噪声带和 `claim` 的最小差距；`accept` 向 `queue.tsv` 入队，队列超限时 `trial --seed 1` 被拒；`claim` 出队并用两 seed 均值回写父本基准；`report` 在每次状态变更后自动重写 `REPORT.md`。

**Tech Stack:** Python 3.10 标准库（`hashlib`、`json`、`math`、`re`、`statistics`、`subprocess`）。测试用 `unittest`，临时 git 仓库 + 假训练命令，不碰 GPU。

**Spec:** `docs/superpowers/specs/2026-10-21-harness-v2-design.md`

- `Z = 2.0`，`R = 5`，`K`（`queue_limit`）`= 2`，`M`（`rotate_after`）`= 8`。全部写进 `run.json`，不做模块级常量。
- `TIMEOUT_SECONDS = 600` 不变。
- `judge.py` 不执行 `git commit`、`git reset`、`git checkout`。只读 git 用 `git_output`。
- 写盘顺序：先 `append_result` + `fsync`，成功后才整文件替换 `parents.tsv` / `queue.tsv` / `REPORT.md`；整文件替换一律临时文件 + `os.replace`。
- 干净工作区检查覆盖四个文件：`train.py`、`prepare.py`、`judge.py`、`task.py`。
- 测试只用标准库，不启动 GPU，不跑真实训练。
- 不做 v1 → v2 记录格式兼容读取。
- `decision` 词表：`基线` / `好于父本` / `分不开` / `差于父本` / `已记录` / `代码坏了` / `测量失败` / `进入名单` / `替换父本` / `未留下` / `成立` / `还判不了` / `更差` / `没有更好`。v1 的 `持平` 删除。

## Review Focus

spec 没有写、但会咬到使用者的五类输入，每条在括号里的 Task 补测试：

1. **`σ = 0`**（`calibrate` 的 R 次分数完全相同）→ `band = 0`，任何微小差异都判 `好于父本`，v2 退化成 v1。应当拒绝写 `run.json` 并提示任务确定性过强。（Task 3）
2. **队列里的提交在 git 里已不可达**（agent reset 掉了它）→ 背压永远解不开，搜索死锁。`trial` 应当自动把不可达的条目丢出队列。（Task 7）
3. **`run.json` 畸形或缺字段**（人手改坏了）→ 命令应当以退出码 2 拒绝，而不是抛 `KeyError` / `JSONDecodeError`。（Task 2）
4. **`parents.tsv` 的 `seeds` 字段为空或畸形** → `parent_n` 除零。应当回落到 `n = 1`。（Task 4）
5. **`DATA_ROOT` 含符号链接或目录项** → `data_manifest` 应当只收普通文件，不跟随符号链接，避免循环与不稳定的清单。（Task 1）

```python
def data_manifest(data_root: Path) -> str | None:
    if not data_root.is_dir():
        return None
    entries = []
    for path in data_root.rglob("*"):
        if path.is_symlink() or not path.is_file():
            continue
        entries.append(
            (path.relative_to(data_root).as_posix(), path.stat().st_size)
        )
    entries.sort(key=lambda item: item[0])
    return "".join(f"{name} {size}\n" for name, size in entries)


def eval_hash(task_py: Path, prepare_py: Path, data_root: Path) -> str | None:
    if not task_py.is_file() or not prepare_py.is_file():
        return None
    manifest = data_manifest(data_root)
    if manifest is None:
        return None
    payload = (
        file_sha256(task_py) + "\n" + file_sha256(prepare_py) + "\n" + manifest
    )
    digest = hashlib.sha256()
    digest.update(payload.encode("utf-8"))
    return digest.hexdigest()
```

```python
RUN_FILE = "run.json"
RUN_FIELDS: tuple[tuple[str, type], ...] = (
    ("run_tag", str),
    ("run_start", str),
    ("baseline_commit", str),
    ("eval_hash", str),
    ("repeats", int),
    ("z", float),
    ("sigma", float),
    ("min_improvement", float),
    ("queue_limit", int),
    ("rotate_after", int),
)


def read_run(repo: Path) -> dict | None:
    path = repo / RUN_FILE
    if not path.is_file():
        return None
    try:
        config = json.loads(path.read_text(encoding="utf-8"))
    except (ValueError, OSError):
        return None
    if not isinstance(config, dict):
        return None
    for name, kind in RUN_FIELDS:
        if name not in config:
            return None
        value = config[name]
        if kind is float and isinstance(value, int) and not isinstance(value, bool):
            config[name] = float(value)
            continue
        if not isinstance(value, kind) or isinstance(value, bool):
            return None
    return config


def write_run(repo: Path, config: dict) -> None:
    path = repo / RUN_FILE
    temporary = path.with_name(path.name + ".tmp")
    with temporary.open("w", encoding="utf-8") as handle:
        json.dump(config, handle, ensure_ascii=False, indent=2, sort_keys=True)
        handle.write("\n")
        handle.flush()
        os.fsync(handle.fileno())
    os.replace(temporary, path)


def band_for(sigma: float, z: float, n: int) -> float:
    count = n if n >= 1 else 1
    return z * sigma * math.sqrt(1.0 + 1.0 / count)
```

```python
def calibration(scores: list[float], z: float) -> tuple[float, float]:
    sigma = statistics.stdev(scores)
    return sigma, z * sigma


def cmd_calibrate(
    repo: Path, command: list[str], timeout: float, test_root: bool
) -> int:
```

```python
def rotation_blocked(
    rows: list[dict], parent_count: int, head_parent: str, rotate_after: int
) -> bool:
    if parent_count < 2 or rotate_after < 1:
        return False
    attempts = [
        row for row in rows

    seen = {row["parent"] for row in window}
```

```python
def read_queue(path: Path) -> list[dict]:
    if not path.is_file():
        return []
    lines = path.read_text(encoding="utf-8").splitlines()
    rows = []
    for line in lines[1:]:
        if line == "":
            continue
        rows.append(dict(zip(QUEUE_FIELDS, line.split("\t"))))
    return rows


def write_queue(path: Path, rows: list[dict]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    with temporary.open("w", encoding="utf-8") as handle:
        handle.write(QUEUE_HEADER + "\n")
        for row in rows:
            handle.write(_line(QUEUE_FIELDS, row) + "\n")
        handle.flush()
        os.fsync(handle.fileno())
    os.replace(temporary, path)


def enqueue(rows: list[dict], entry: dict) -> list[dict]:
    if any(row["commit"] == entry["commit"] for row in rows):
        return list(rows)
    return list(rows) + [entry]


def dequeue(rows: list[dict], commit: str) -> list[dict]:
    return [row for row in rows if row["commit"] != commit]

    kept = []
    for row in rows:
        if git_output(repo, "cat-file", "-e", row["commit"] + "^{commit}") is None:
            continue
        kept.append(row)
    return kept
```

```python
def claim_verdict(
    candidate_seed1: float | None,
    candidate_seed2: float | None,
    parent_seed1: float | None,
    parent_seed2: float | None,
    eval_hashes: list[str],
    min_improvement: float | None,
    n: int,
) -> tuple[str, str]:
    if (
        len(eval_hashes) != 4
        or len(set(eval_hashes)) != 1
        or any(item == "" for item in eval_hashes)
    ):
        return "还判不了", "尺子或数据不同"
    missing = []
    if candidate_seed1 is None:
        missing.append("候选种子1")
    if candidate_seed2 is None:
        missing.append("候选种子2")
    if parent_seed1 is None:
        missing.append("父本种子1")
    if parent_seed2 is None:
        missing.append("父本种子2")
    if missing:
        return "还判不了", "缺" + "、".join(missing)
    if min_improvement is None or min_improvement <= 0:
        return "还判不了", "未标定"
    improvement_1 = parent_seed1 - candidate_seed1
    improvement_2 = parent_seed2 - candidate_seed2
    if improvement_1 == improvement_2 and improvement_1 != 0:
        return "还判不了", "两次改进没有分开"
    average = (improvement_1 + improvement_2) / 2.0
    if average < 0:
        return "更差", ""
    penalty = luck_penalty(candidate_seed1, candidate_seed2, n)
    if average > min_improvement and average - penalty > min_improvement:
        return "成立", ""
    return "没有更好", ""


FINAL_CLAIMS = frozenset({"成立", "更差", "没有更好"})


def apply_claim(
    parents: list[dict], commit: str, decision: str, mean: float
) -> list[dict]:
    if decision not in FINAL_CLAIMS:
        return list(parents)
    if not any(row["commit"] == commit for row in parents):
        return list(parents)
    if decision == "更差" and len(parents) > 1:
        return [row for row in parents if row["commit"] != commit]
    updated = []
    for row in parents:
        if row["commit"] != commit:
            updated.append(row)
            continue
        fresh = dict(row)
        fresh["reviewed"] = "yes"
        if decision != "更差":
            fresh["val_bpb"] = f"{mean:.6f}"
            fresh["seeds"] = "1,2"
        updated.append(fresh)
    return updated
```

```python
def build_report(
    config: dict | None,
    rows: list[dict],
    parents: list[dict],
    queue: list[dict],
) -> str:
    out: list[str] = ["# 搜索报告", ""]
    out.append("## 标定")
    out.append("")
    if config is None:
        out.append("未标定。先运行 `uv run judge.py calibrate`。")
    else:
        out.append(f"- run_tag：{config['run_tag']}")
        out.append(f"- run_start：{config['run_start']}")
        out.append(f"- baseline：{config['baseline_commit'][:7]}")
        out.append(f"- R = {config['repeats']}，Z = {config['z']}")
        out.append(f"- σ = {config['sigma']:.6f}")
        out.append(f"- band（新父本）= {band_for(config['sigma'], config['z'], 1):.6f}")
        out.append(f"- min_improvement = {config['min_improvement']:.6f}")
        out.append(f"- queue_limit = {config['queue_limit']}，rotate_after = {config['rotate_after']}")
    out.append("")
    out.append("## 轮次统计")
    out.append("")
    groups: dict[str, dict[str, int]] = {}
    for row in rows:
        key = row["eval_hash"]
        groups.setdefault(key, {})
        groups[key][row["decision"]] = groups[key].get(row["decision"], 0) + 1
    if not groups:
        out.append("还没有记录。")
    for key in sorted(groups):
        out.append(f"### 尺子 {key[:12]}")
        out.append("")
        out.append("| 判定 | 次数 |")
        out.append("|---|---|")
        for decision in sorted(groups[key]):
            out.append(f"| {decision} | {groups[key][decision]} |")
        out.append("")
    out.append("## 搜索结果（未验证）")
    out.append("")
    if not parents:
        out.append("名单为空。")
    else:
        out.append("| commit | 机制 | val_bpb | seeds | 已复核 |")
        out.append("|---|---|---|---|---|")
        for row in parents:
            out.append(
                f"| {row['commit'][:7]} | {row['mechanism']} | {row['val_bpb']} "
                f"| {row['seeds']} | {row['reviewed']} |"
            )
    out.append("")
    out.append("## 已验证结论")
    out.append("")
    holds = [row for row in rows if row["decision"] == "成立"]
    if not holds:
        out.append("无。进入名单不是已验证结论。")
    else:
        out.append("| commit | 机制 |")
        out.append("|---|---|")
        for row in holds:
            out.append(f"| {row['commit'][:7]} | {row['mechanism']} |")
    out.append("")
    out.append("## 未复核警告")
    out.append("")
    if not queue:
        out.append("队列为空。")
    else:
        out.append(f"还有 {len(queue)} 条待复核：")
        for row in queue:
            out.append(f"- {row['commit'][:7]}（{row['mechanism']}，{row['accepted_at']}）")
    out.append("")
    out.append("## 机制审计")
    out.append("")
    knobs_by_mechanism: dict[str, set[str]] = {}
    for row in rows:
        if row["seed"] not in {"1", "2"} or row["mechanism"] == "基线":
            continue
        knobs_by_mechanism.setdefault(row["mechanism"], set()).add(row["knobs"])
    flagged = {
        name: values for name, values in knobs_by_mechanism.items() if len(values) > 1
    }
    if not flagged:
        out.append("同名机制的旋钮声明一致。")
    else:
```

````markdown
## The experiment loop

The first command on a clean baseline commit is:

```text
uv run judge.py calibrate

improvement.

LOOP FOREVER:

1. Read `parents.tsv`, `results.tsv`, and `queue.tsv`. If the queue is

uv run judge.py trial --seed 1 --mechanism 学习率 --knobs LEARNING_RATE \

```

6. `好于父本` and the complexity is worth it → `uv run judge.py accept`.
   `分不开` and the code is simpler → `uv run judge.py accept --simpler
   --replace <parent-hash>`. `差于父本`, or not worth it → `git reset`
   back to the parent.
7. When the queue is not empty, pick an entry and review it: check out
````
